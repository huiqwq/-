void poly_multiply(int *a, int *b, int *ab, int n, int p){
    for(int i = 0; i < n; ++i){
        for(int j = 0; j < n; ++j){
            ab[i+j]=(1LL * a[i] * b[j] % p + ab[i+j]) % p;
        }
    }
}

long long qpow(long long a, long long b, int p) {
    long long res = 1;
    while (b) {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}

inline u64 mul_mod(u64 a, u64 b, u64 p) {
    return (u64)((u128)a * b % p);
}

u64 qpow64(u64 a, u64 b, u64 p) {
    u64 res = 1;
    while (b) {
        if (b & 1) res = mul_mod(res, a, p);
        a = mul_mod(a, a, p);
        b >>= 1;
    }
    return res;
}

struct Montgomery32{
    u32 mod;
    u32 inv;
    u32 r2;
    explicit Montgomery32(u32 p) : mod(p) {
        u32 x = 1;
        for (int i = 0; i < 5; ++i) {
            x *= 2 - mod * x;
        }
        inv = 0 - x;

        u64 r = (u64(1) << 32) % mod;
        r2 = u32(r * r % mod);
    }
    u32 reduce(u64 x) const {
    //    u32 q = u32(x) * u64(inv);
        u32 q = (u32)((x & 0xffffffffULL) * inv);
        u64 y = (x + u64(q) * mod) >> 32;
        if (y >= mod) y -= mod;
        return u32(y);
    }

    u32 init(u32 x) const {
        return reduce(u64(x) * r2);
    }
        u32 value(u32 x) const {
        return reduce(x);
    }

    u32 one() const {
        return init(1);
    }

    u32 add(u32 a, u32 b) const {
        u32 c = a + b;
        if (c >= mod) c -= mod;
        return c;
    }

    u32 sub(u32 a, u32 b) const {
        return a >= b ? a - b : a + mod - b;
    }

    u32 mul(u32 a, u32 b) const {
        return reduce(u64(a) * b);
    }
};

struct Montgomery64 {
    u64 mod;
    u64 inv;
    u64 r2;

    explicit Montgomery64(u64 p) : mod(p) {
        u64 x = 1;
        for (int i = 0; i < 6; ++i) {
            x *= 2 - mod * x;
        }
        inv = 0 - x;
        u64 r = (u64)((u128(1) << 64) % mod);
        r2 = (u64)((u128)r * r % mod);
    }

    u64 reduce(u128 x) const {
        u64 q = (u64)x * inv;
        u128 y = (x + (u128)q * mod) >> 64;
        if (y >= mod) y -= mod;
        return (u64)y;
    }

    u64 init(u64 x) const {
        return reduce((u128)x * r2);
    }

    u64 value(u64 x) const {
        return reduce(x);
    }

    u64 one() const {
        return init(1);
    }

    u64 add(u64 a, u64 b) const {
        u64 c = a + b;
        if (c >= mod || c < a) c -= mod;
        return c;
    }

    u64 sub(u64 a, u64 b) const {
        return a >= b ? a - b : a + mod - b;
    }

    u64 mul(u64 a, u64 b) const {
        return reduce((u128)a * b);
    }
};

static inline uint32x2_t montgomery_reduce2_neon(uint64x2_t x, uint32_t mod, uint32_t inv) {
    uint32x2_t low = vmovn_u64(x);
    uint32x2_t q = vmul_u32(low, vdup_n_u32(inv));
    uint64x2_t q_mod = vmull_u32(q, vdup_n_u32(mod));
    uint64x2_t y64 = vshrq_n_u64(vaddq_u64(x, q_mod), 32);
    uint32x2_t y = vmovn_u64(y64);
    uint32x2_t mod_v = vdup_n_u32(mod);
    uint32x2_t ge_mod = vcge_u32(y, mod_v);
    return vsub_u32(y, vand_u32(ge_mod, mod_v));
}

static inline uint32x4_t montgomery_mul4_neon(uint32x4_t a, uint32x4_t b, const Montgomery32& mont) {
    uint64x2_t prod_low = vmull_u32(vget_low_u32(a), vget_low_u32(b));
    uint64x2_t prod_high = vmull_u32(vget_high_u32(a), vget_high_u32(b));
    uint32x2_t low = montgomery_reduce2_neon(prod_low, mont.mod, mont.inv);
    uint32x2_t high = montgomery_reduce2_neon(prod_high, mont.mod, mont.inv);
    return vcombine_u32(low, high);
}

static inline void butterfly_addsub4_neon(int *left, int *right, uint32x4_t v, uint32_t mod) {
    uint32x4_t mod_v = vdupq_n_u32(mod);
    uint32x4_t u = vreinterpretq_u32_s32(vld1q_s32(reinterpret_cast<const int32_t*>(left)));

    uint32x4_t sum = vaddq_u32(u, v);
    uint32x4_t sum_ge_mod = vcgeq_u32(sum, mod_v);
    sum = vsubq_u32(sum, vandq_u32(sum_ge_mod, mod_v));

    uint32x4_t diff = vsubq_u32(u, v);
    uint32x4_t diff_need_mod = vcltq_u32(u, v);
    diff = vaddq_u32(diff, vandq_u32(diff_need_mod, mod_v));

    vst1q_s32(reinterpret_cast<int32_t*>(left), vreinterpretq_s32_u32(sum));
    vst1q_s32(reinterpret_cast<int32_t*>(right), vreinterpretq_s32_u32(diff));
}

static inline void butterfly_dif4_neon(int *left, int *right, uint32x4_t roots, const Montgomery32& mont) {
    uint32x4_t mod_v = vdupq_n_u32(mont.mod);
    uint32x4_t u = vreinterpretq_u32_s32(vld1q_s32(reinterpret_cast<const int32_t*>(left)));
    uint32x4_t v = vreinterpretq_u32_s32(vld1q_s32(reinterpret_cast<const int32_t*>(right)));

    uint32x4_t sum = vaddq_u32(u, v);
    uint32x4_t sum_ge_mod = vcgeq_u32(sum, mod_v);
    sum = vsubq_u32(sum, vandq_u32(sum_ge_mod, mod_v));

    uint32x4_t diff = vsubq_u32(u, v);
    uint32x4_t diff_need_mod = vcltq_u32(u, v);
    diff = vaddq_u32(diff, vandq_u32(diff_need_mod, mod_v));

    uint32x4_t scaled_diff = montgomery_mul4_neon(diff, roots, mont);

    vst1q_s32(reinterpret_cast<int32_t*>(left), vreinterpretq_s32_u32(sum));
    vst1q_s32(reinterpret_cast<int32_t*>(right), vreinterpretq_s32_u32(scaled_diff));
}

static inline void butterfly_addsub2_neon_u64(u64 *left, u64 *right, const u64 v_buf[2], u64 mod) {
    uint64x2_t mod_v = vdupq_n_u64(mod);
    uint64x2_t u = vld1q_u64(left);
    uint64x2_t v = vld1q_u64(v_buf);

    uint64x2_t sum = vaddq_u64(u, v);
    uint64x2_t sum_ge_mod = vcgeq_u64(sum, mod_v);
    sum = vsubq_u64(sum, vandq_u64(sum_ge_mod, mod_v));

    uint64x2_t diff = vsubq_u64(u, v);
    uint64x2_t diff_need_mod = vcltq_u64(u, v);
    diff = vaddq_u64(diff, vandq_u64(diff_need_mod, mod_v));

    vst1q_u64(left, sum);
    vst1q_u64(right, diff);
}

void ntt_dif_forward_simd(std::vector<int>& f, int p) {
    int n = f.size();
    Montgomery32 mont((uint32_t)p);

    for (int len = n; len >= 2; len >>= 1) {
        uint32_t wn = mont.init((uint32_t)qpow(3, (p - 1) / len, p));
        int half = len >> 1;

        for (int i = 0; i < n; i += len) {
            uint32_t w = mont.one();
            int j = 0;
            for (; j + 4 <= half; j += 4) {
                uint32_t roots_buf[4];
                roots_buf[0] = w;
                roots_buf[1] = mont.mul(roots_buf[0], wn);
                roots_buf[2] = mont.mul(roots_buf[1], wn);
                roots_buf[3] = mont.mul(roots_buf[2], wn);
                w = mont.mul(roots_buf[3], wn);

                uint32x4_t roots = vld1q_u32(roots_buf);
                butterfly_dif4_neon(&f[i + j], &f[i + j + half], roots, mont);
            }
            for (; j < half; ++j) {
                uint32_t u = (uint32_t)f[i + j];
                uint32_t v = (uint32_t)f[i + j + half];

                f[i + j] = (int)mont.add(u, v);
                f[i + j + half] = (int)mont.mul(mont.sub(u, v), w);

                w = mont.mul(w, wn);
            }
        }
    }
}

void ntt_dit_inverse_simd(std::vector<int>& f, int p) {
    int n = f.size();
    Montgomery32 mont((uint32_t)p);

    for (int len = 2; len <= n; len <<= 1) {
        int wn_normal = (int)qpow(3, (p - 1) / len, p);
        wn_normal = (int)qpow(wn_normal, p - 2, p);
        uint32_t wn = mont.init((uint32_t)wn_normal);
        int half = len >> 1;

        for (int i = 0; i < n; i += len) {
            uint32_t w = mont.one();
            int j = 0;

            for (; j + 4 <= half; j += 4) {
                uint32_t roots_buf[4];
                roots_buf[0] = w;
                roots_buf[1] = mont.mul(roots_buf[0], wn);
                roots_buf[2] = mont.mul(roots_buf[1], wn);
                roots_buf[3] = mont.mul(roots_buf[2], wn);
                w = mont.mul(roots_buf[3], wn);

                uint32x4_t roots = vld1q_u32(roots_buf);
                uint32x4_t right = vreinterpretq_u32_s32(
                    vld1q_s32(reinterpret_cast<const int32_t*>(&f[i + j + half]))
                );
                uint32x4_t v = montgomery_mul4_neon(roots, right, mont);

                butterfly_addsub4_neon(&f[i + j], &f[i + j + half], v, (uint32_t)p);
            }

            for (; j < half; ++j) {
                uint32_t u = (uint32_t)f[i + j];
                uint32_t v = mont.mul(w, (uint32_t)f[i + j + half]);

                f[i + j] = (int)mont.add(u, v);
                f[i + j + half] = (int)mont.sub(u, v);

                w = mont.mul(w, wn);
            }
        }
    }

    uint32_t inv_n = mont.init((uint32_t)qpow(n, p - 2, p));
    uint32x4_t inv_n_vec = vdupq_n_u32(inv_n);
    int i = 0;
    for (; i + 4 <= n; i += 4) {
        uint32x4_t val = vreinterpretq_u32_s32(vld1q_s32(reinterpret_cast<const int32_t*>(&f[i])));
        uint32x4_t scaled = montgomery_mul4_neon(val, inv_n_vec, mont);
        vst1q_s32(reinterpret_cast<int32_t*>(&f[i]), vreinterpretq_s32_u32(scaled));
    }
    for (; i < n; ++i) {
        f[i] = (int)mont.mul((uint32_t)f[i], inv_n);
    }
}

void poly_multiply_ntt_ditdif_simd(int *a, int *b, int *ab, int n, int p) {
    int lim = 1;
    while (lim < 2 * n - 1) lim <<= 1;

    Montgomery32 mont((uint32_t)p);
    std::vector<int> A(lim), B(lim);

    for (int i = 0; i < n; ++i) {
        A[i] = (int)mont.init((uint32_t)a[i]);
        B[i] = (int)mont.init((uint32_t)b[i]);
    }

    ntt_dif_forward_simd(A, p);
    ntt_dif_forward_simd(B, p);

    int i = 0;
    for (; i + 4 <= lim; i += 4) {
        uint32x4_t va = vreinterpretq_u32_s32(vld1q_s32(reinterpret_cast<const int32_t*>(&A[i])));
        uint32x4_t vb = vreinterpretq_u32_s32(vld1q_s32(reinterpret_cast<const int32_t*>(&B[i])));
        uint32x4_t vc = montgomery_mul4_neon(va, vb, mont);
        vst1q_s32(reinterpret_cast<int32_t*>(&A[i]), vreinterpretq_s32_u32(vc));
    }
    for (; i < lim; ++i) {
        A[i] = (int)mont.mul((uint32_t)A[i], (uint32_t)B[i]);
    }

    ntt_dit_inverse_simd(A, p);

    for (int i = 0; i < 2 * n - 1; ++i) {
        ab[i] = (int)mont.value((uint32_t)A[i]);
    }
}

void ntt_u64_simd(std::vector<u64>& f, u64 p, bool inverse) {
    int n = f.size();
    Montgomery64 mont(p);

    for (int i = 1, j = 0; i < n; ++i) {
        int bit = n >> 1;
        while (j & bit) {
            j ^= bit;
            bit >>= 1;
        }
        j ^= bit;
        if (i < j) std::swap(f[i], f[j]);
    }

    for (int len = 2; len <= n; len <<= 1) {
        u64 wn_normal = qpow64(3, (p - 1) / (u64)len, p);
        if (inverse) {
            wn_normal = qpow64(wn_normal, p - 2, p);
        }
        u64 wn = mont.init(wn_normal);
        int half = len >> 1;

        for (int i = 0; i < n; i += len) {
            u64 w = mont.one();
            int j = 0;

            for (; j + 2 <= half; j += 2) {
                u64 v_buf[2];
                v_buf[0] = mont.mul(w, f[i + j + half]);
                w = mont.mul(w, wn);
                v_buf[1] = mont.mul(w, f[i + j + half + 1]);
                w = mont.mul(w, wn);

                butterfly_addsub2_neon_u64(&f[i + j], &f[i + j + half], v_buf, p);
            }

            for (; j < half; ++j) {
                u64 u = f[i + j];
                u64 v = mont.mul(w, f[i + j + half]);

                f[i + j] = mont.add(u, v);
                f[i + j + half] = mont.sub(u, v);

                w = mont.mul(w, wn);
            }
        }
    }

    if (inverse) {
        u64 inv_n = mont.init(qpow64((u64)n, p - 2, p));
        for (int i = 0; i < n; ++i) {
            f[i] = mont.mul(f[i], inv_n);
        }
    }
}

void poly_multiply_ntt_u64_simd(u64 *a, u64 *b, u64 *ab, int n, u64 p) {
    int lim = 1;
    while (lim < 2 * n - 1) lim <<= 1;

    Montgomery64 mont(p);
    std::vector<u64> A(lim), B(lim);

    for (int i = 0; i < n; ++i) {
        A[i] = mont.init(a[i]);
        B[i] = mont.init(b[i]);
    }

    ntt_u64_simd(A, p, false);
    ntt_u64_simd(B, p, false);

    for (int i = 0; i < lim; ++i) {
        A[i] = mont.mul(A[i], B[i]);
    }

    ntt_u64_simd(A, p, true);

    for (int i = 0; i < 2 * n - 1; ++i) {
        ab[i] = mont.value(A[i]);
    }
}
